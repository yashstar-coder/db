CREATE TABLE Borrower (
    Rollin NUMBER PRIMARY KEY,
    Name VARCHAR2(50),
    DateofIssue DATE,
    NameofBook VARCHAR2(100),
    Status CHAR(1) -- 'I' for Issued, 'R' for Returned
);

CREATE TABLE Fine (
    Roll_no NUMBER,
    Fine_Date DATE,
    Amt NUMBER
);


INSERT INTO Borrower VALUES (1, 'Raj', SYSDATE - 20, 'DBMS Handbook', 'I');
INSERT INTO Borrower VALUES (2, 'Anjali', SYSDATE - 40, 'Operating Systems', 'I');
COMMIT;

SET SERVEROUTPUT ON;
SET VERIFY OFF;

DECLARE
    v_roll      Borrower.Rollin%TYPE := &Enter_Roll_No;
    v_book      Borrower.NameofBook%TYPE := '&Enter_Book_Name';
    v_issue_dt  DATE;
    v_days      NUMBER;
    v_fine      NUMBER := 0;

BEGIN
    
    SELECT DateofIssue INTO v_issue_dt 
    FROM Borrower 
    WHERE Rollin = v_roll AND NameofBook = v_book AND Status = 'I';

   
    v_days := TRUNC(SYSDATE) - TRUNC(v_issue_dt);
    DBMS_OUTPUT.PUT_LINE('Total Days: ' || v_days);

    
    IF v_days > 15 AND v_days <= 30 THEN
        v_fine := (v_days - 15) * 5;
        
    ELSIF v_days > 30 THEN
        -- First 15 days of fine (from day 16 to 30) at Rs 5
        -- Remaining days (after 30) at Rs 50
        v_fine := (15 * 5) + ((v_days - 30) * 50);
    END IF;

 
    UPDATE Borrower 
    SET Status = 'R' 
    WHERE Rollin = v_roll AND NameofBook = v_book;

   
    IF v_fine > 0 THEN
        INSERT INTO Fine (Roll_no, Fine_Date, Amt)
        VALUES (v_roll, SYSDATE, v_fine);
        DBMS_OUTPUT.PUT_LINE('Fine Amount Calculated: Rs. ' || v_fine);
    ELSE
        DBMS_OUTPUT.PUT_LINE('No fine applicable. Book returned on time.');
    END IF;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Transaction completed and record updated to Returned (R).');

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Error: No active issued record found for this Roll No and Book.');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('An unexpected error occurred: ' || SQLERRM);
END;
/
