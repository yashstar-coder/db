CREATE TABLE Library (
    Book_ID NUMBER PRIMARY KEY,
    Title VARCHAR2(100),
    Author VARCHAR2(100),
    Price NUMBER(8, 2)
);


CREATE TABLE Library_Audit (
    Audit_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Book_ID NUMBER,
    Old_Title VARCHAR2(100),
    Old_Author VARCHAR2(100),
    Old_Price NUMBER(8, 2),
    Operation_Type VARCHAR2(10), 
    Changed_By VARCHAR2(50),
    Changed_At TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE OR REPLACE TRIGGER trg_library_audit
BEFORE UPDATE OR DELETE ON Library
FOR EACH ROW
DECLARE
    v_op VARCHAR2(10);
BEGIN

    IF UPDATING THEN
        v_op := 'UPDATE';
    ELSIF DELETING THEN
        v_op := 'DELETE';
    END IF;

   
    INSERT INTO Library_Audit (
        Book_ID, 
        Old_Title, 
        Old_Author, 
        Old_Price, 
        Operation_Type, 
        Changed_By
    )
    VALUES (
        :OLD.Book_ID, 
        :OLD.Title, 
        :OLD.Author, 
        :OLD.Price, 
        v_op, 
        USER
    );
END;
/

INSERT INTO Library VALUES (101, 'The Great Gatsby', 'F. Scott Fitzgerald', 15.00);
COMMIT;


UPDATE Library SET Price = 12.50 WHERE Book_ID = 101;


DELETE FROM Library WHERE Book_ID = 101;


SELECT Book_ID, Old_Price, Operation_Type, Changed_At FROM Library_Audit;
