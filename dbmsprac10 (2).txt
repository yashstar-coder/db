db.Books.aggregate([
{
    $group: {

        total_books: { $sum: 1 }

    $group: {

    }
}
])

{
    $group: {
        _id: null,
        maximum_price: { $max: "$price" },
        minimum_price: { $min: "$price" }
    }
}
])


db.Books.aggregate([
{
    $match: {
        publish_year: { $gt: 2015 }

{
