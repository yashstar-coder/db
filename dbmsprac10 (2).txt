// ======================================================
// 2. Create an Index on price field in Products Collection
// ======================================================

db.Products.createIndex({ price: 1 })

// ======================================================
// 3. Create Compound Index on author_name and publish_year
// ======================================================

db.Books.createIndex({ author_name: 1, publish_year: 1 })

// ======================================================
// 4. View All Indexes in Books Collection
// ======================================================

db.Books.getIndexes()

// ======================================================
// 5. View All Indexes in Products Collection
// ======================================================

db.Products.getIndexes()

// ======================================================
// 6. Drop Index Created on price field in Products
// ======================================================

// First check index name
db.Products.getIndexes()

// Drop index
db.Products.dropIndex({ price: 1 })

// ======================================================
// 7. Total Number of Books Published by Each Author
// ======================================================

db.Books.aggregate([
{
    $group: {
        _id: "$author_name",
        total_books: { $sum: 1 }
    }
}
])

// ======================================================
// 8. Calculate Average Price of All Products
// ======================================================

db.Products.aggregate([
{
    $group: {
        _id: null,
        average_price: { $avg: "$price" }
    }
}
])

// ======================================================
// 9. Find Maximum and Minimum Price in Products
// ======================================================

db.Products.aggregate([
{
    $group: {
        _id: null,
        maximum_price: { $max: "$price" },
        minimum_price: { $min: "$price" }
    }
}
])

// ======================================================
// 10. Count Books Where publish_year > 2015
// ======================================================

db.Books.aggregate([
{
    $match: {
        publish_year: { $gt: 2015 }
    }
},
{
    $count: "Total_Books"
}
])

// ======================================================
// 11. Display Only book_title and author_name
// ======================================================

db.Books.aggregate([
{
    $project: {
        _id: 0,
        book_title: 1,
        author_name: 1
    }
}
])

// ======================================================
// 12. Sort Documents in Books Collection
// ======================================================

// Sort by publish_year ascending
db.Books.find().sort({ publish_year: 1 })

// Sort by publish_year descending
db.Books.find().sort({ publish_year: -1 })
